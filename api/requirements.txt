# Serving dependencies for the TranslateGemma API.
#
# Standalone on purpose: api/ is deployed without the rest of the repository, and
# this installs only what inference needs -- not the training/evaluation stack
# (trl, datasets, unbabel-comet, liger-kernel, MetricX).
#
# The model-side versions are the ones in the project's uv.lock, so the served
# model behaves like the one the evaluation harness scored. Update them together.

# Same CUDA wheel line as the trainer image. Requires
# UV_INDEX_STRATEGY=unsafe-best-match (set in the Dockerfile), or pip, so that
# `torch` resolves against this index rather than stopping at PyPI.
--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.8.0

transformers==4.57.6
tokenizers==0.22.2
peft==0.20.0
huggingface-hub==0.36.2
safetensors==0.8.0
sentencepiece==0.2.2
pillow==12.3.0
# transformers 4.57 is not compatible with the NumPy 2 ABI across all of its
# optional paths; the project lock stays on the 1.26 line.
numpy==1.26.4

# Optional 4-bit serving (TG_LOAD_IN_4BIT=true). Versions below 0.48 lack
# Blackwell (sm_120) kernels.
bitsandbytes==0.50.0

# Optional sentence splitting (TG_SPLIT_SENTENCES=true).
pysbd==0.3.4

# The web stack is range-pinned rather than exact-pinned: unlike the packages
# above there is no lockfile entry to copy, and these ranges are the
# compatibility boundaries that matter (Pydantic v2 API, FastAPI's lifespan
# handler). For a byte-reproducible rebuild, copy
# /opt/resolved-requirements.txt out of a built image and pin from that.
fastapi[standard]>=0.115,<1.0
uvicorn>=0.34,<1.0
pydantic>=2.9,<3.0
pydantic-settings>=2.5,<3.0
