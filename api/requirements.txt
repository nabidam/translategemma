# Serving dependencies for the TranslateGemma API gateway.
#
# Standalone on purpose: api/ is deployed without the rest of the repository.
# The weights are served by vLLM in its own container, so nothing here pulls
# torch, peft or bitsandbytes -- this process renders prompts and forwards
# them over HTTP, and the image is a few hundred MB rather than several GB.
#
# transformers/tokenizers are still exact-pinned to the project's uv.lock
# versions: they produce the prompt token ids vLLM generates from, so a
# tokenizer that drifts from the evaluated one changes the served translation.

transformers==4.57.6
tokenizers==0.22.2
huggingface-hub==0.36.2
sentencepiece==0.2.2
# Imported by the multimodal processor class TranslateGemma's tokenizer ships
# inside, even though no image ever reaches it.
pillow==12.3.0
# transformers 4.57 is not compatible with the NumPy 2 ABI across all of its
# optional paths; the project lock stays on the 1.26 line.
numpy==1.26.4

# Sentence splitting: optional for the server (TG_SPLIT_SENTENCES=true),
# required by scripts/benchmark_speed.py, which segments a benchmarked page the
# same way the server would so the two report the same segment count.
pysbd==0.3.4

# Used only by scripts/benchmark_speed.py, to pull a page of text out of a PDF.
# Installed rather than left optional because the deployment hosts run offline,
# where `pip install` at benchmark time is not an option. Never imported by the
# server. Range-pinned: it is not in the project lockfile.
pymupdf>=1.24,<2

# The web stack is range-pinned rather than exact-pinned: unlike the packages
# above there is no lockfile entry to copy, and these ranges are the
# compatibility boundaries that matter (Pydantic v2 API, FastAPI's lifespan
# handler). For a byte-reproducible rebuild, copy
# /opt/resolved-requirements.txt out of a built image and pin from that.
fastapi[standard]>=0.115,<1.0
uvicorn>=0.34,<1.0
pydantic>=2.9,<3.0
pydantic-settings>=2.5,<3.0
# The upstream client. Comes with fastapi[standard], but this module imports it
# directly, so it is declared directly.
httpx>=0.27,<1.0
